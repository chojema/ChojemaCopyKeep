cmake_minimum_required(VERSION 3.24)
project(ChojemaCopyKeep VERSION 0.4.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "This project currently builds the Windows version only.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ChojemaCopyKeep WIN32
  src/main.cpp src/store.cpp src/image.cpp src/update.cpp resources/app.rc)
target_compile_definitions(ChojemaCopyKeep PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN COPYKEEP_VERSION="${PROJECT_VERSION}")
target_include_directories(ChojemaCopyKeep PRIVATE src resources)
target_link_libraries(ChojemaCopyKeep PRIVATE winsqlite3 bcrypt gdiplus comctl32 shell32 ole32 uuid shlwapi dwmapi uxtheme winhttp imm32 oleaut32)
if(MSVC)
  set_property(TARGET ChojemaCopyKeep PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ChojemaCopyKeep PRIVATE /utf-8 /W4 /permissive- /EHsc)
  target_link_options(ChojemaCopyKeep PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(replacement_ui_tests tests/replacement_ui_test.cpp)
  target_include_directories(replacement_ui_tests PRIVATE src)
  target_compile_definitions(replacement_ui_tests PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(replacement_ui_tests PRIVATE ole32 oleaut32 uuid imm32)
  add_test(NAME replacement_selection COMMAND replacement_ui_tests)
  set_tests_properties(replacement_selection PROPERTIES TIMEOUT 30)
  add_executable(replacement_tests tests/replacement_test.cpp)
  target_include_directories(replacement_tests PRIVATE src)
  if(MSVC)
    target_compile_options(replacement_tests PRIVATE /utf-8)
    target_compile_options(replacement_ui_tests PRIVATE /utf-8)
  endif()
  add_test(NAME replacement_matching COMMAND replacement_tests)
  add_executable(update_tests tests/update_test.cpp)
  target_include_directories(update_tests PRIVATE src)
  add_test(NAME update_versions COMMAND update_tests)
  add_test(NAME storage_and_images COMMAND ChojemaCopyKeep --self-test ${CMAKE_BINARY_DIR}/test-data)
  add_test(NAME popup_korean COMMAND ChojemaCopyKeep --smoke-test ko ${CMAKE_BINARY_DIR}/test-ui/preview-ko.png)
  add_test(NAME popup_english COMMAND ChojemaCopyKeep --smoke-test en ${CMAKE_BINARY_DIR}/test-ui/preview-en.png)
  add_test(NAME settings_korean COMMAND ChojemaCopyKeep --smoke-test ko ${CMAKE_BINARY_DIR}/test-ui/settings-ko.png settings)
  add_test(NAME settings_english COMMAND ChojemaCopyKeep --smoke-test en ${CMAKE_BINARY_DIR}/test-ui/settings-en.png settings)
  add_test(NAME snippets_korean COMMAND ChojemaCopyKeep --smoke-test ko ${CMAKE_BINARY_DIR}/test-ui/snippets-ko.png snippets)
  add_test(NAME snippets_english COMMAND ChojemaCopyKeep --smoke-test en ${CMAKE_BINARY_DIR}/test-ui/snippets-en.png snippets)
  add_test(NAME snippet_editor_korean COMMAND ChojemaCopyKeep --smoke-test ko ${CMAKE_BINARY_DIR}/test-ui/snippet-editor-ko.png snippet-editor)
  add_test(NAME snippet_editor_english COMMAND ChojemaCopyKeep --smoke-test en ${CMAKE_BINARY_DIR}/test-ui/snippet-editor-en.png snippet-editor)
  set_tests_properties(popup_korean popup_english settings_korean settings_english
    snippets_korean snippets_english snippet_editor_korean snippet_editor_english PROPERTIES TIMEOUT 60)
endif()
install(TARGETS ChojemaCopyKeep RUNTIME DESTINATION .)
install(FILES README.md README.en.md LICENSE DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md")
